\begin{algorithm}[H]
\caption{Example Sorting Algorithm}
\Input{List of integers $A[1 \dots n]$}
\Output{Sorted list}
\For{$i \gets 1$ \KwTo $n$}{
    \For{$j \gets 1$ \KwTo $n - i$}{
        \If{$A[j] > A[j+1]$}{
            Swap $A[j]$ and $A[j+1]$\;
        }
    }
}
\end{algorithm}
